function parseClock(t) {
  const m = String(t || '').match(/(\d{1,2}):(\d{2})\s*([AaPp])/); if (!m) return null;
  let h = Number(m[1]) % 12; if (m[3].toUpperCase() === 'P') h += 12; return h * 60 + Number(m[2]);
}
function fmt(min) { const h = Math.floor(min / 60), m = min % 60; return ((h % 12) || 12) + ':' + (m < 10 ? '0' : '') + m + ' ' + (h < 12 ? 'AM' : 'PM'); }
let EVENT = null;
function nowInfo() {
  // Returns minutes since midnight if today is the event day, otherwise null
  if (BOOT.now) return parseClock(BOOT.now);
  if (!EVENT || !EVENT.iso) return null;
  const d = new Date();
  const iso = d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
  return iso === EVENT.iso ? d.getHours() * 60 + d.getMinutes() : null;
}

function leader(code) {
  spin();
  call('apiLeader', String(code || '').trim()).then(d => {
    if (!d.ok) {
      if (d.volunteer) return volunteer(code);
      if (!EVENT) call('apiHome').then(h => paintHeader(h.event));
      if (d.inactive) return paintInactive();
      return codeForm(d.locked ? 'Too many wrong codes were tried. Please wait 15 minutes.' : 'That link or code didn’t work. It may have been replaced with a new one. Ask your keyman or the overseers for your current link.');
    }
    L = d; L.code = String(code).trim().toUpperCase(); LINKS = d.links || {};
    SEEN_MEM = null; SEEN_REL = null;   // reload what this leader has read from the server
    applySharedNotes(L);
    if (SHARED_NOTES === null) loadSharedNotes().then(() => { if (L && SHARED_NOTES) { applySharedNotes(L); paintLeader(); } });
    paintHeader(d.event);
    setUrl({ code: L.code });
    if (UI.slot === null || UI.slot >= d.slots.length) UI.slot = currentSlot();
    if (UI.sec !== 'ALL' && d.sections.indexOf(UI.sec) < 0) UI.sec = 'ALL';
    paintLeader();
    clearInterval(refreshTimer);
    refreshTimer = setInterval(() => silentRefresh(), 120000);
  }, e => showError(e.message));
}
function silentRefresh() {
  if (!L || PENDING) return;   // a save is still going; refresh once it finishes
  const seq = EDITSEQ;
  call('apiLeader', L.code).then(d => {
    if (!d.ok) return;
    whenStill(() => {
      if (seq !== EDITSEQ || PENDING) return;   // an edit happened meanwhile; this copy is already out of date
      const c = L.code; L = d; L.code = c; LINKS = d.links || {}; applySharedNotes(L); repaintInPlace();
    });
  });
}

// Wait until no finger is on the screen and scrolling has stopped. Redrawing mid-swipe
// swaps the grid out from under the finger, which can leave the page stuck until you swipe sideways.
let TOUCHING = false, LASTSCROLL = 0;
document.addEventListener('touchstart', () => { TOUCHING = true; }, { passive: true });
document.addEventListener('touchend', () => { TOUCHING = false; }, { passive: true });

  fn();
}

// Apply a swap/fill/remove to the page's own copy of the schedule (same rules as the server)
function applyEditLocal(r, k, oldName, newName) {
  const now = r.cells[k] || [];
  let names = now.filter(n => !oldName || norm(n) !== norm(oldName));
  if (newName) { const at = now.map(norm).indexOf(norm(oldName)); if (oldName && at >= 0) names.splice(Math.min(at, names.length), 0, newName); else names.push(newName); }
  r.cells[k] = names;
  r.allDay = false;   // the server writes an all-day row out slot by slot
  const tag = r.spot + ' · ' + r.section, sl = L.slots[k];
  if (oldName) {
    const rp = rosterOf(oldName);
    if (rp && rp.busy[k] != null) { const left = String(rp.busy[k]).split(' + ').filter(x => x !== tag); if (left.length) rp.busy[k] = left.join(' + '); else delete rp.busy[k]; }
    const pp = (L.people || []).filter(p => norm(p.name) === norm(oldName))[0];
    if (pp) pp.blocks = pp.blocks.map(b => Object.assign({}, b, { cells: (b.cells || []).filter(c => !(c[0] === r.id && c[1] === k)) })).filter(b => b.cells.length);
    if (pp && !pp.blocks.length) L.people = L.people.filter(p => p !== pp);
  }
  if (newName) {
    let rp = rosterOf(newName);
    if (rp) rp.busy[k] = rp.busy[k] != null ? rp.busy[k] + ' + ' + tag : tag;
    let pp = (L.people || []).filter(p => norm(p.name) === norm(newName))[0];
    if (!pp) { pp = { name: newName, blocks: [] }; L.people.push(pp); L.people.sort((a, b) => a.name.toLowerCase() < b.name.toLowerCase() ? -1 : 1); }
    pp.blocks.push({ timeText: r.override || (sl.label + (sl.end != null ? ' – ' + fmt(sl.end) : '')), spot: r.spot, section: r.section, startMin: sl.start, endMin: sl.end, cells: [[r.id, k]] });
    pp.blocks.sort((a, b) => a.startMin - b.startMin);
  }
}
function currentSlot() {
  const now = nowInfo();
  if (now === null || !L) return 0;
  let idx = 0;
  L.slots.forEach((s, i) => { if (now >= s.start) idx = i; });
  return idx;
}
function liveSlot() {
  const now = nowInfo(); if (now === null || !L) return -1;
  let idx = -1; L.slots.forEach((s, i) => { if (now >= s.start) idx = i; }); return idx;
}

/* ---------- "Read" tracking for Conflicts and What's new ----------
   Saved on the server for each leader (L.seen), because browsers inside Apps Script
   can lose their saved data. This phone/computer's storage is only a backup. */
let SEEN_MEM = null;   // short ids of the conflicts already seen
function saveSeen(data) { call('apiMarkSeen', L.code, data).catch(() => {}); }
function seenId(k) { let h = 5381; for (let i = 0; i < k.length; i++) h = ((h << 5) + h + k.charCodeAt(i)) | 0; return (h >>> 0).toString(36); }
function conflictKeys() {
  const k = [];
  (L.issues || []).forEach(i => { if (!(i.type === 'Double-booked' && i.ignored)) k.push(issueKey(i)); });
  (L.cancelled || []).forEach(c => k.push(cancelKey(c)));
  return k;
}
function issueKey(i) { return 'i|' + i.type + '|' + (i.key || i.text); }
function cancelKey(c) { return 'c|' + norm(c.name) + '|' + c.when; }
function seenSet() {
  if (!SEEN_MEM) {
    let local = [];
    try { local = JSON.parse(localStorage.getItem('attSeenConflicts') || '[]'); } catch (e) {}
    // older versions saved full conflict keys here; turn them into short ids
    local = (Array.isArray(local) ? local : []).map(k => /^[ic]\|/.test(k) ? seenId(k) : String(k));
    const server = (L.seen && L.seen.conflicts) || null;
    SEEN_MEM = server ? server.concat(local.filter(x => server.indexOf(x) < 0)) : local;
    if (!server && local.length) saveSeen({ conflicts: SEEN_MEM });   // first time: copy this browser's list to the server
  }
  return SEEN_MEM;
}
function markConflictsSeen(keys) {
  const ids = keys.map(seenId);
  const same = ids.length === seenSet().length && ids.every(x => SEEN_MEM.indexOf(x) >= 0);
  SEEN_MEM = ids;
  try { localStorage.setItem('attSeenConflicts', JSON.stringify(ids)); } catch (e) {}
  if (!same) saveSeen({ conflicts: ids });
}

/* ---------- Shared release notes ----------
   One notes file on the shared page serves every copy. If it can't be reached, the page keeps the notes from this
   schedule's own Release Notes tab. Overseers also learn when their copy's server code is older than the newest. */
const NOTES_URL = 'https://volunteer-schedule.github.io/releases.json';
let SHARED_NOTES = null;   // null = not tried yet, false = not available, else the file
function loadSharedNotes() {
  if (SHARED_NOTES !== null) return Promise.resolve(SHARED_NOTES);
  return fetch(BOOT.shared ? 'releases.json' : NOTES_URL, { cache: 'no-cache' })
    .then(r => (r.ok ? r.json() : null)).catch(() => null)
    .then(j => (SHARED_NOTES = j && Array.isArray(j.releases) ? j : false));
}
function verNewer(a, b) {   // is version a newer than b? ("1.10" > "1.9")
  const x = String(a || '0').split('.').map(n => parseInt(n, 10) || 0), y = String(b || '0').split('.').map(n => parseInt(n, 10) || 0);
  for (let i = 0; i < Math.max(x.length, y.length); i++) if ((x[i] || 0) !== (y[i] || 0)) return (x[i] || 0) > (y[i] || 0);
  return false;
}
function applySharedNotes(d) {
  if (!d || !SHARED_NOTES) return;
  d.releases = SHARED_NOTES.releases.map(r => ({ version: r.version, date: r.date || '', title: r.title || '',
    items: (r.items || []).filter(i => d.me.all || i.audience !== 'Overseers').map(i => ({ type: i.type, note: i.note })) }))
    .filter(r => r.items.length);
  d.updateAvailable = d.me.all && verNewer(SHARED_NOTES.latestServer, d.appVersion) ? SHARED_NOTES.latestServer : '';
}
function updateNotice(d) {
  if (!d.updateAvailable) return '';
  return '<div class="note" style="margin-top:10px">⬆ A newer version of the schedule app is ready for your copy (yours: ' + esc(d.appVersion || 'older') +

}

// Redraw the leader page without losing your place: the page and the Full day grid stay scrolled where they were
function paintLeader() {
  const g = $('.gridwrap'), gx = g ? g.scrollLeft : 0, gy = g ? g.scrollTop : 0, y = window.scrollY, tab = UI.tab, mode = UI.mode;
  drawLeader();
  if (UI.tab !== tab || UI.mode !== mode) return;   // switched tabs or views: start fresh
  const n = $('.gridwrap');
  if (n && g) { n.scrollLeft = gx; n.scrollTop = gy; }
  window.scrollTo(0, y);
}
function drawLeader() {
  const d = L, me = d.me;
  const issues = d.issues || [];
  const dups = issues.filter(i => i.type === 'Double-booked' && !i.ignored).length;

function rowsInView() { return L.rows.filter(r => UI.sec === 'ALL' || r.section === UI.sec); }
// open = nobody there, not marked "No shift needed", and not an early slot like 6:45 / 7:15
// (unless someone was removed or cancelled from that early spot)
function isOpenCell(r, i) {
  return ((L.noOpenSlots || []).indexOf(i) < 0 || (L.vacated || []).some(v => v[0] === r.id && v[1] === i)) && !(r.cells[i] || []).length && (r.closed || []).indexOf(i) < 0 && !unschedAt(r, i) && !reducedAt(r, i);
}
// spot cut back for low traffic at slot i (only while nobody is in it, "No shift needed" or not)
function reducedAt(r, i) {
  if (!r || !L.slots[i] || (r.cells[i] || []).length) return null;
  return (L.reduced || []).filter(u => norm(u.section) === norm(r.section) && norm(u.pos) === norm(r.pos) && u.time === L.slots[i].label)[0] || null;
}
// "Leave unscheduled" entry covering this empty spot at slot i (only while it's still empty)
function unschedAt(r, i) {
  if ((r.cells[i] || []).length || (r.closed || []).indexOf(i) >= 0) return null;
  return (L.unscheduled || []).filter(u => norm(u.section) === norm(r.section) && norm(u.pos) === norm(r.pos) &&
    (u.time === 'All day' || u.time === L.slots[i].label)).sort((a, b) => (a.time === 'All day') - (b.time === 'All day'))[0] || null;
}
// open spots always show; this red bar just says how many there are
function openBar(n, what) {
  return '<div class="openbar"><span class="grow">⚠ Open spots<span class="sub">' + (n === 1 ? '1 spot needs' : n + ' spots need') + ' to be filled ' + what + '</span></span>' +
    '<span class="cnt">' + n + '</span></div>';
}

// Label for a time slot: use the Time Override (e.g. "12:00 PM until baptism ends") when every spot shown in it has the same one
function slotLabel(i, rows) {
  const on = rows.filter(r => (r.cells[i] || []).length);
  if (on.length && on.every(r => r.override && r.override === on[0].override)) return on[0].override;
  return L.slots[i].label;
}

// Keymen: the same shift instructions their volunteers see, one tap-to-open box per kind of spot.
// Instructions already shown on the keyman's own shifts aren't repeated here.
function teamInstructions(d) {
  const mine = (d.myBlocks || []).map(b => b.instructions).filter(Boolean);
  const byText = {};
  (d.rows || []).forEach(r => {
    if (!r.instructions || mine.indexOf(r.instructions) >= 0) return;
    const e = byText[r.instructions] = byText[r.instructions] || { spots: [], text: r.instructions };
    if (e.spots.indexOf(r.spot) < 0) e.spots.push(r.spot);
  });
  const list = Object.keys(byText).map(k => byText[k]);
  if (!list.length) return '';
  return '<div class="card"><p class="eyebrow">Shift instructions for your team</p><p class="small muted" style="margin:0 0 6px">What your volunteers see on these spots.</p>' +
    list.map(e => '<details class="instr"><summary>📋 ' + esc(e.spots.join(', ')) + '</summary><div>' + esc(e.text) + '</div></details>').join('') + '</div>';
}

// Overseers: set the morning/afternoon count times on the day. Nobody sees a count until its time is set.
function countEditor(counts) {
  const val = c => { if (c.min == null) return ''; const h = Math.floor(c.min / 60), m = c.min % 60; return (h < 10 ? '0' : '') + h + ':' + (m < 10 ? '0' : '') + m; };
  const summary = counts.map(c => (c.key === 'am' ? 'AM ' : 'PM ') + (c.min == null ? 'not set' : c.time) +
    (c.reported ? ' – ' + c.total.toLocaleString() + ' (' + c.reported + '/' + c.tally.length + ')' : '')).join(' · ');
  const head = '<button class="collhead" id="cttoggle" aria-expanded="' + !!UI.ctOpen + '"><span class="grow"><span class="eyebrow" style="margin:0">Attendance counts</span>' +
    '<span class="small muted" style="display:block">' + esc(summary) + '</span></span><span class="plus">' + (UI.ctOpen ? '−' : '+') + '</span></button>';
  if (!UI.ctOpen) return '<div class="card">' + head + '</div>';
  return '<div class="card">' + head + '<p class="small muted" style="margin:10px 0 10px">Set each time when you know it. Only spots ' + esc(L.countSpotsText || 'listed under Attendance Count Spots') + ' on duty at that time and their keymen will see it. Until a time is set, no one sees it. Once it’s set, each counter types their spot’s total on their page and it shows below.</p>' +
    counts.map(c => '<div class="countrow"><label class="grow" for="ct-' + c.key + '"><strong>' + (c.key === 'am' ? 'Morning' : 'Afternoon') + '</strong><div class="small muted">' + (c.min == null ? 'Not set · hidden' : 'Showing') + '</div></label>' +
      '<input type="time" id="ct-' + c.key + '" value="' + esc(UI.ct && UI.ct[c.key] != null ? UI.ct[c.key] : val(c)) + '">' +
      (c.min == null ? '' : '<button class="btn sm line" data-ctclear="' + c.key + '">Clear</button>') + '</div>').join('') +
    '<button class="btn" id="ctsave" style="width:100%;margin-top:10px">Save count times</button>' +
    counts.filter(c => c.min != null || c.reported).map(c => '<div class="tallylist"><h4>' + (c.key === 'am' ? 'Morning' : 'Afternoon') + ' count</h4>' +
      '<p class="small muted" style="margin:0 0 4px">' + c.reported + ' of ' + c.tally.length + ' in. Tap a spot to fix its number or text the brother counting it.</p>' +
      c.tally.map(t => '<button class="tallyrow" data-tfix="' + c.key + '|' + esc(t.spot) + '"><span><strong>' + esc(t.spot) + '</strong>' +
        (t.by ? '<span class="small muted" style="display:block">' + esc(firstLast(t.by)) + ' · ' + esc(t.when.split(' ').slice(1).join(' ')) + '</span>' :
          t.area ? '<span class="small muted" style="display:block">Keyman’s count</span>' : '') + '</span>' +
        (t.n != null ? '<span class="n">' + t.n.toLocaleString() + '</span>' : '<span class="n miss">not in yet</span>') + '</button>').join('') +
      '<div class="tallytotal"><span>Total</span><span>' + c.total.toLocaleString() + '</span></div></div>').join('') + '</div>';
}
// Who is responsible for this count, with a Text button that has a reminder ready to send
function countWho(t, ck) {
  const who = t.who || [];
  if (!who.length) return '<p class="small muted" style="margin:14px 0 0">' + (t.area ? 'No keyman is listed for this area.' : 'No one is scheduled at this spot at the count time.') + '</p>';
  const c = (L.counts || []).filter(x => x.key === ck)[0] || {};
  const msg = p => 'Hi ' + firstLast(p.name).split(' ')[0] + ', please send in the ' + (ck === 'am' ? 'morning' : 'afternoon') + ' attendance count for ' + t.spot +
    (c.time ? ' (' + c.time + ' count)' : '') + '. You can enter it on your schedule page. Thanks!';
  return '<p class="eyebrow" style="margin:16px 0 6px">' + (t.area ? 'Keyman counting this area' : 'Counting at ' + esc(c.time || 'the count time')) + '</p>' + who.map(p =>
    '<div class="contact"><div class="avatar">' + esc(initials(p.name)) + '</div><div class="grow" style="font-weight:600">' + esc(firstLast(p.name)) + '</div>' +
    (p.phone ? '<div class="btns"><a class="btn sm" href="' + smsHref(p.phone, t.n == null ? msg(p) : '') + '">💬 Text</a><a class="btn sm ghost" href="' + telHref(p.phone) + '">Call</a></div>' : '<span class="small muted">No number</span>') + '</div>').join('');
}
// Overseers: type in or fix one spot's number (e.g. reported by radio)
function openTallyFix(id) {
  const ck = id.slice(0, id.indexOf('|')), spot = id.slice(id.indexOf('|') + 1);
  const t = ((L.counts || []).filter(c => c.key === ck)[0].tally || []).filter(x => x.spot === spot)[0] || {};
  openSheet('<div class="big">' + esc(spot) + ' · ' + (ck === 'am' ? 'Morning' : 'Afternoon') + ' count</div>' +
    (t.by ? '<p class="small muted">Last entered by ' + esc(firstLast(t.by)) + ' at ' + esc(t.when) + '</p>' : '<p class="small muted">Not in yet.</p>') +
    '<input class="tallynum" type="number" inputmode="numeric" min="0" id="tfixn" value="' + (t.n != null ? t.n : '') + '" placeholder="Total" style="width:100%;margin:6px 0 12px">' +
    '<button class="btn" id="tfixsave" style="width:100%">Save</button>' +
    (t.n != null ? '<button class="btn line" id="tfixclear" style="width:100%;margin-top:8px">Clear this count</button>' : '') + countWho(t, ck));
  const save = (n, btn) => {
    if (n !== '' && !/^\d+$/.test(n)) { toast('Type the count as a number'); return; }
    busy(btn, 'Saving…');
    call('apiEdit', L.code, { tally: true, count: ck, spot: spot, n: n }).then(res => {
      if (!res.ok) { openSheet('<div class="big">Not saved</div><p>' + esc(res.error) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'); return; }
      closeSheet(); toast(n === '' ? 'Count cleared' : 'Count saved'); silentRefresh();
    }, e => openSheet('<div class="big">Not saved</div><p>' + esc(e.message) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'));
  };
  $('#tfixsave').onclick = () => save($('#tfixn').value.trim(), $('#tfixsave'));
  if ($('#tfixclear')) $('#tfixclear').onclick = () => save('', $('#tfixclear'));
}
function bindCountEditor() {
  if ($('#cttoggle')) $('#cttoggle').onclick = () => { UI.ctOpen = !UI.ctOpen; paintLeader(); };
  document.querySelectorAll('[data-tfix]').forEach(b => b.onclick = () => openTallyFix(b.dataset.tfix));
  if (!$('#ctsave')) return;
  const toText = v => { if (!v) return ''; const [h, m] = v.split(':').map(Number); return fmt(h * 60 + m); };
  const save = (req, btn) => {
    busy(btn, 'Saving…');
    call('apiEdit', L.code, Object.assign({ counts: true }, req)).then(res => {
      if (!res.ok) { openSheet('<div class="big">Not saved</div><p>' + esc(res.error) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'); silentRefresh(); return; }
      UI.ct = {}; toast('Count times saved'); silentRefresh();
    }, e => openSheet('<div class="big">Not saved</div><p>' + esc(e.message) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'));
  };
  // keep what's typed if the page auto-refreshes before Save
  ['am', 'pm'].forEach(k => { const i = $('#ct-' + k); if (i) i.oninput = () => { UI.ct = UI.ct || {}; UI.ct[k] = i.value; }; });
  $('#ctsave').onclick = () => save({ am: toText($('#ct-am').value), pm: toText($('#ct-pm').value) }, $('#ctsave'));
  document.querySelectorAll('[data-ctclear]').forEach(b => b.onclick = () => { const r = {}; r[b.dataset.ctclear] = ''; save(r, b); });
}

function paintSchedule() {
  const d = L, live = liveSlot();
  let h = '';
  if (d.sections.length > 1) {
    h += '<div class="chips">' + ['ALL'].concat(d.sections).map(s => '<button class="chip ' + (UI.sec === s ? 'on' : '') + '" data-sec="' + esc(s) + '">' +
      (s === 'ALL' ? 'All sections' : esc(s)) + '</button>').join('') + '</div>';
  }
  const seg = (k, label) => '<button data-mode="' + k + '" class="' + (UI.mode === k ? 'on' : '') + '">' + label + '</button>';
  h += '<div class="row" style="margin-bottom:10px"><div class="grow"></div><div class="seg">' + seg('time', 'By time') + seg('grid', 'Full day') + seg('person', 'By person') + '</div></div>';

  // Only spots that have someone scheduled, and only the time slots that are used (someone on duty, or a spot to fill)
  const vr = rowsInView().filter(r => r.cells.some(c => c.length));
  const used = d.slots.map((s, i) => i).filter(i => vr.some(r => (r.cells[i] || []).length) || rowsInView().some(r => isOpenCell(r, i)));

  if (UI.mode === 'person') {
    h += '<input type="search" id="pq" placeholder="Type a name…" autocomplete="off" value="' + esc(UI.pq || '') + '"><div id="plist" style="margin-top:12px"></div>';
  } else if (!used.length) {
    h += '<div class="card center muted">Nothing is scheduled here yet.</div>';
  } else {
    if (used.indexOf(UI.slot) < 0) {
      const pick = used.filter(i => i <= Math.max(live, 0)).pop();
      UI.slot = pick != null ? pick : used[0];
    }
    if (UI.mode === 'time') {
      h += '<div class="slots">' + used.map(i => '<button class="slot ' + (UI.slot === i ? 'on' : '') + '" data-slot="' + i + '">' +
        (i === live ? '<small>NOW</small>' : '') + esc(slotLabel(i, vr)) + '</button>').join('') + '</div>';
      const k = UI.slot, sl = d.slots[k], lab = slotLabel(k, vr);
      const nk = used[used.indexOf(k) + 1];
      h += lab !== sl.label
        ? '<p class="small muted" style="margin:0 0 10px">Showing who is on duty <strong>' + esc(lab) + '</strong>.</p>'
        : '<p class="small muted" style="margin:0 0 10px">Showing who is on duty from <strong>' + esc(sl.label) + '</strong> to <strong>' + esc(sl.end != null ? fmt(sl.end) : 'end of program') + '</strong>.</p>';
      const openRows = rowsInView().filter(r => isOpenCell(r, k));
      if (openRows.length) h += openBar(openRows.length, 'at this time');
      // keep the schedule's own order, with open spots in their place (e.g. 3B right after 3)
      const bySec = {};
      rowsInView().filter(r => (r.cells[k] || []).length || isOpenCell(r, k) || unschedAt(r, k) || reducedAt(r, k))
        .forEach(r => (bySec[r.section] = bySec[r.section] || []).push(r));
      Object.keys(bySec).forEach(sec => {
        const leads = (d.sectionLeads[sec] || []).map(firstLast).join(', ');
        h += '<div class="card"><div class="sec-head"><h3>' + esc(sec) + '</h3><span class="lead">' + (leads ? 'Keyman: ' + esc(leads) : '') + '</span></div>' +
          (d.sectionNotes[sec] || []).map(n =>
            n.kind === 'alert' ? (d.me.all ? alertBox([n]) : '') :
            n.kind === 'time' ? (n.time === sl.start ? '<div class="note" style="margin:0 0 8px">📝 ' + esc(n.text) + '</div>' : '') :
            '<div class="note" style="margin:0 0 8px">' + esc(n.text) + '</div>').join('');
        bySec[sec].forEach(r => {
          const names = r.cells[k] || [];
          if (reducedAt(r, k)) {
            h += '<div class="spotrow unsched" data-red="' + r.id + '" data-rslot="' + k + '"><div class="sp">' + esc(r.spot) + '</div><div><span class="redpill">↓ Reduced Staffing-Low Traffic</span>' +
              '<span class="unschednote">Not scheduled because traffic is low. The brother before leaves at the end of his shift.</span></div></div>';
            return;
          }
          const un = unschedAt(r, k);
          if (un) {
            h += '<div class="spotrow unsched" data-un="' + r.id + '" data-uslot="' + k + '"><div class="sp">' + esc(r.spot) + '</div><div><span class="unschedpill">⊘ Not scheduled</span>' +
              '<span class="unschednote">ⓘ ' + esc(un.note) + '</span></div></div>';
            return;
          }
          if (!names.length) {
            h += '<div class="spotrow openspot"><div class="sp">' + esc(r.spot) + '</div><div class="who"><span class="openpill">⚠ OPEN</span>' +
              '<button class="btn sm fill" data-fill="' + r.id + '" data-fslot="' + k + '">+ Fill</button><button class="btn sm ign" data-ign-open="' + r.id + '" data-islot="' + k + '">Ignore</button></div></div>';
            return;
          }
          const next = nk != null ? (r.cells[nk] || []) : [];
          const changes = next.length && next.join('|') !== names.join('|');
          h += '<div class="spotrow"><div class="sp">' + esc(r.spot) + (r.override && lab === sl.label ? '<small>' + esc(r.override) + '</small>' : (r.allDay ? '<small>All day</small>' : '')) + '</div><div class="who">' +
            names.map(n => personBtn(n, r, k)).join('') +
            (reducedAt(r, k + 1) ? '<div class="nextup">↓ Reduced after this shift (low traffic). No relief, can leave at ' + esc(fmt(L.slots[k + 1].start)) + '.</div>' :
             changes ? '<div class="nextup">Next (' + esc(slotLabel(nk, vr)) + '): ' + next.map(n => esc(firstLast(n))).join(', ') + '</div>' : '') + '</div></div>';
        });
        // spots with "No shift needed" on the sheet at this time: closed until tapped (the list can be long),
        // then tap a spot to see why or reopen it. Stays open or closed through refreshes.
        const cl = rowsInView().filter(r => r.section === sec && closedAt(r, k));
        const clKey = sec + '|' + k;
        if (cl.length) h += '<details class="closedline" data-clkey="' + esc(clKey) + '"' + ((UI.clOpen || {})[clKey] ? ' open' : '') + '><summary>ⓘ No shift needed (' + cl.length + ')</summary><div class="cllist">' +

        h += '</div>';
      });
    } else {
      let openCount = 0;
      rowsInView().forEach(r => used.forEach(i => { if (isOpenCell(r, i)) openCount++; }));
      if (openCount) h += openBar(openCount, 'today');
      const gr = rowsInView().filter(r => r.cells.some(c => c.length) || used.some(i => isOpenCell(r, i) || unschedAt(r, i) || reducedAt(r, i) || closedAt(r, i)));
      h += '<div class="gridwrap"><table class="grid"><thead><tr><th class="first">Spot</th>' + used.map(i => '<th class="' + (i === live ? 'nowcol' : '') + '">' +
        esc(slotLabel(i, vr)) + (i === live ? ' ●' : '') + '</th>').join('') + '</tr></thead><tbody>';
      let lastSec = null;
      gr.forEach(r => {
        if (r.section !== lastSec) { lastSec = r.section; h += '<tr class="sec"><td colspan="' + (used.length + 1) + '"><span>' + esc(r.section) + '</span></td></tr>'; }

// a person's assignments today, e.g. "7:45 AM · Spot 3"
function todayLines(rp) {
  // a spot with a Time Override (e.g. baptism "12:00 PM until baptism ends") shows that time instead of the column's
  const override = i => (L.rows.filter(r => r.override && (r.cells[i] || []).some(n => norm(n) === norm(rp.name)))[0] || {}).override;
  return rp ? Object.keys(rp.busy).map(Number).sort((a, b) => a - b).map(i => (override(i) || L.slots[i].label) + ' · ' + rp.busy[i]) : [];
}

// overseers: anyone but themselves; keymen: people in their own sections (their roster)

}

/* ---------- swap / fill picker ---------- */
function restInfo(p, k) {
  const busy = Object.keys(p.busy).map(Number).filter(i => i !== k);
  const prev = busy.filter(i => i < k).sort((a, b) => b - a)[0];
  const next = busy.filter(i => i > k).sort((a, b) => a - b)[0];
  const sl = L.slots[k];
  const prevEnd = prev != null ? (L.slots[prev].end != null ? L.slots[prev].end : L.slots[prev].start + 60) : null;
  const slotEnd = sl.end != null ? sl.end : sl.start + 60;
  return {
    busyNow: p.busy[k] != null ? p.busy[k] : null,
    before: prevEnd == null ? Infinity : sl.start - prevEnd,
    after: next == null ? Infinity : L.slots[next].start - slotEnd,
    prevEnd: prevEnd, nextStart: next != null ? L.slots[next].start : null
  };
}
function dur(m) { if (m === Infinity) return ''; const h = Math.floor(m / 60), mm = m % 60; return h ? h + ' hr' + (mm ? ' ' + mm + ' min' : '') : mm + ' min'; }

function openPicker(ctx, oldName) {
  const r = rowById(ctx.rowId), k = ctx.slot;
  if (!r) return;
  const hidden = L.hiddenSections || [];
  const rowHidden = hidden.indexOf(norm(r.section)) >= 0;
  const inCell = (r.cells[k] || []).map(norm);
  const list = (L.roster || []).filter(p => p.pickable && inCell.indexOf(norm(p.name)) < 0 && (rowHidden || !p.blocked))
    .map(p => Object.assign({ p: p }, restInfo(p, k)))
    .sort((a, b) => (a.busyNow ? 1 : 0) - (b.busyNow ? 1 : 0) || b.before - a.before || b.after - a.after || (a.p.name < b.p.name ? -1 : 1));
  const title = oldName ? 'Replace ' + firstLast(oldName) : 'Fill this spot';
  openSheet('<div class="big">' + esc(title) + '</div><div class="muted small" style="margin:4px 0 12px">' + esc(r.spot) + ' · ' + esc(r.section) + ' · ' + esc(slotLabel(k, [r])) +
    (r.allDay ? ' (this time slot only)' : '') + '</div>' +
    (oldName ? '' : '<button class="btn sm ign" id="pkign" style="margin:0 0 10px">⊘ Ignore — leave this spot unscheduled</button>') +
    '<input type="search" id="kq" placeholder="Search ' + (L.me.all ? 'everyone' : 'your team') + '…" autocomplete="off"><div id="krows" style="margin-top:8px"></div>' +
    '<div class="pickfoot"><button class="btn line" style="width:100%" onclick="closeSheet()">Cancel</button></div>', 'fixedlist');
  const paint = () => {
    const q = norm($('#kq').value);
    $('#krows').scrollTop = 0;
    const shown = list.filter(x => !q || norm(x.p.name).includes(q) || norm(firstLast(x.p.name)).includes(q)).slice(0, 60);
    $('#krows').innerHTML = shown.map((x, i) => {
      const status = x.busyNow ? '<span class="warn">⚠ Already at ' + esc(x.busyNow) + ' this time</span>'
        : (x.before === Infinity ? 'No shift before this' : 'Off since ' + fmt(x.prevEnd) + ' (' + dur(x.before) + ')') +
          ' · ' + (x.nextStart == null ? 'no later shifts' : 'next at ' + fmt(x.nextStart));
      return '<button class="cand' + (x.busyNow ? ' busy' : '') + '" data-i="' + list.indexOf(x) + '"><div class="grow"><strong>' + esc(firstLast(x.p.name)) + '</strong>' +
        badges(x.p) + '<div class="small muted">' + status + '</div>' +
        (x.p.limits ? '<div class="limits"><span>ⓘ <strong>Limitations:</strong> ' + esc(x.p.limits) + '</span></div>' : '') + '</div><span class="chev">›</span></button>';
    }).join('') || '<p class="muted">No one matches that name.</p>';
    $('#krows').querySelectorAll('.cand').forEach(b => b.onclick = () => { const x = list[Number(b.dataset.i)]; confirmEdit(ctx, oldName, x.p.name, x.busyNow); });
  };
  $('#kq').oninput = paint; paint();
  if ($('#pkign')) $('#pkign').onclick = () => confirmUnsched(r, k);
}

/* ---------- open spots left unscheduled on purpose ---------- */
// Only spots with their own number can be reduced (rows like Exteriors share one name, so they can't be told apart)
function ownSpot(r) { return r.pos && L.rows.filter(x => x.section === r.section && x.pos === r.pos).length === 1; }
// "Ignore" on an open spot: pick which of this spot's open times, then a reason (or Reduced Staffing-Low Traffic)

  if (!r) return;
  const open = L.slots.map((sl, j) => j).filter(j => isOpenCell(r, j));
  const canReduce = ownSpot(r);
  openSheet('<div class="big">Leave ' + esc(r.spot) + ' unscheduled?</div><p class="muted small" style="margin:4px 0 12px">' + esc(r.section) + '</p>' +
    '<p style="margin:0 0 10px">These times <strong>won’t be scheduled</strong> and won’t show as open spots. Everyone will see the reason instead.</p>' +
    '<p class="eyebrow" style="margin:12px 0 6px">For which times?</p>' +
    open.map(j => '<label class="radio"><input type="checkbox" name="utime" value="' + j + '"' + (j === k ? ' checked' : '') + '> ' + esc(slotLabel(j, [r])) + '</label>').join('') +
    '<p class="eyebrow" style="margin:14px 0 6px">Reason (everyone can see it)</p>' +
    (canReduce ? '<button class="btn line" id="ured" aria-pressed="false" style="width:100%;margin-bottom:8px">↓ Reduced Staffing-Low Traffic</button>' : '') +

// "No shift needed" typed on the Schedule sheet: show why, and offer to open it (which erases that text on the sheet)
function closedAt(r, i) { return !(r.cells[i] || []).length && (r.closed || []).indexOf(i) >= 0 && !reducedAt(r, i); }
function closedInfo(r, k) {
  if (!r || !closedAt(r, k)) return;
  openSheet('<div class="big">ⓘ ' + esc(r.spot) + ' · not scheduled</div><p class="muted small" style="margin:4px 0 12px">' + esc(r.section) + ' · ' + esc(slotLabel(k, [r])) + '</p>' +
    '<div class="note" style="margin:0 0 10px">ⓘ “No shift needed” was typed in this spot on the Google Sheet schedule.</div>' +
    '<p class="small muted" style="margin:0 0 14px">Reopening erases “No shift needed” from that cell, so the spot shows + Fill.</p>' +
    '<button class="btn line" id="clopen" style="width:100%">Reopen ' + esc(slotLabel(k, [r])) + ' for Scheduling</button><button class="btn" style="width:100%;margin-top:8px" onclick="closeSheet()">Close</button>');
  $('#clopen').onclick = () => reopenThenFill(r, k, { reopen: true, rowId: r.id, section: r.section, pos: r.pos, time: L.slots[k].label },
    () => { r.closed = (r.closed || []).filter(x => x !== k); });
}
function reducedInfo(r, k) {
  const u = reducedAt(r, k);
  if (!u) return;

  openSheet('<div class="big">↓ ' + esc(r.spot) + ' · Reduced Staffing-Low Traffic</div><p class="muted small" style="margin:4px 0 12px">' + esc(r.section) + ' · ' + esc(slotLabel(k, [r])) + '</p>' +
    '<div class="note" style="margin:0 0 10px">No one is scheduled here at this time because traffic is low.' + (before ? ' ' + esc(before) + ' has been told he can leave at ' + esc(fmt(L.slots[k].start)) + ' without waiting for relief.' : '') + '</div>' +

    '<button class="btn line" id="unred" style="width:100%">Reopen ' + esc(slotLabel(k, [r])) + ' for Scheduling</button><button class="btn" style="width:100%;margin-top:8px" onclick="closeSheet()">Close</button>');

  });

// Reopen one time and go straight to picking who fills it; the reopen saves in the background
// (if it can't save, the spot is put back the way it was)
function reopenThenFill(r, k, req, open) {
  const before = { closed: (r.closed || []).slice(), unscheduled: (L.unscheduled || []).slice(), reduced: (L.reduced || []).slice() };
  open();
  beginSave();
  paintLeader();
  openPicker({ rowId: r.id, slot: k }, '');
  const fail = msg => { r.closed = before.closed; L.unscheduled = before.unscheduled; L.reduced = before.reduced; paintLeader(); notSaved(msg); endSave(); };
  call('apiEdit', L.code, req).then(res => { if (!res.ok) fail(res.error); else endSave(); }, e => fail(e.message || 'Check your signal and try again.'));
}
function saveReduced(req, doneMsg) {
  call('apiEdit', L.code, Object.assign({ reduced: true }, req)).then(res => {
    if (!res.ok) { openSheet('<div class="big">Not saved</div><p>' + esc(res.error) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'); if (res.changed) silentRefresh(); return; }
    closeSheet(); toast(doneMsg); silentRefresh();
  }, e => openSheet('<div class="big">Not saved</div><p>' + esc(e.message) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'));
}
function saveUnsched(req, doneMsg) {
  call('apiEdit', L.code, Object.assign({ unscheduled: true }, req)).then(res => {
    if (!res.ok) { openSheet('<div class="big">Not saved</div><p>' + esc(res.error) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'); if (res.changed) silentRefresh(); return; }
    closeSheet(); toast(doneMsg); silentRefresh();
  }, e => openSheet('<div class="big">Not saved</div><p>' + esc(e.message) + '</p><button class="btn" style="width:100%" onclick="closeSheet()">OK</button>'));
}

function confirmEdit(ctx, oldName, newName, busyNow) {
  const r = rowById(ctx.rowId), k = ctx.slot, when = slotLabel(k, [r]);
  const what = oldName && newName ? 'Swap <strong>' + esc(firstLast(oldName)) + '</strong> for <strong>' + esc(firstLast(newName)) + '</strong>'
    : newName ? 'Put <strong>' + esc(firstLast(newName)) + '</strong> in this spot' : 'Remove <strong>' + esc(firstLast(oldName)) + '</strong> and leave this spot open';
  openSheet('<div class="big">Confirm change</div><p style="margin:10px 0 4px">' + what + '</p><p class="muted small" style="margin:0 0 12px">' + esc(r.spot) + ' · ' + esc(r.section) + ' · ' + esc(when) +
    (r.allDay ? ' (this time slot only)' : '') + '</p>' + (newName ? limitsBox(rosterOf(newName)) : '') +
    (busyNow ? '<div class="note" style="border-color:var(--warn);background:var(--warn-soft);color:#7A1A14;margin-bottom:12px">⚠ ' + esc(firstLast(newName)) +
      ' is already scheduled at ' + esc(busyNow) + ' at this time. Save anyway only if you know that’s right.</div>' : '') +
    alsoBox(r, k, oldName, newName) +
    '<div class="btns"><button class="btn' + (newName ? '' : ' danger') + '" id="dosave">' + (busyNow ? 'Save anyway' : newName ? 'Save change' : 'Remove') + '</button><button class="btn line" onclick="closeSheet()">Cancel</button></div>');
  if ($('#alsoall')) $('#alsoall').onclick = () => document.querySelectorAll('[data-also]').forEach(x => { if (!x.dataset.busy) x.checked = true; });
  $('#dosave').onclick = () => {
    // later open times in this spot the leader also wants filled with the same person
    const also = Array.from(document.querySelectorAll('[data-also]')).filter(x => x.checked).map(x => Number(x.dataset.also));
    const busyAt = j => { const rp = rosterOf(newName); return rp && rp.busy[j] != null ? rp.busy[j] : ''; };
    const alsoWarn = {};
    also.forEach(j => { if (busyAt(j)) alsoWarn[j] = firstLast(newName) + ' also at ' + busyAt(j); });
    const req = { rowId: r.id, slot: k, pos: r.pos, section: r.section, expected: (r.cells[k] || []).slice(), oldName: oldName || '', newName: newName || '',
      warning: busyNow ? firstLast(newName) + ' also at ' + busyNow : '', alsoSlots: also, alsoWarn: alsoWarn };
    // show the change right away, then save it in the background
    const snap = snapL(), tok = ++SAVETOK;
    applyEditLocal(r, k, oldName, newName);
    also.forEach(j => applyEditLocal(r, j, '', newName));
    beginSave();
    paintLeader();
    afterEdit({ spot: r.spot, time: [k].concat(also).map(j => r.override || L.slots[j].label).join(', '), oldName: oldName || '', newName: newName || '' }, tok, true);
    call('apiEdit', L.code, req).then(res => {
      if (!res.ok) { restoreL(snap); paintLeader(); notSaved(res.error); endSave(); return; }
      if (sheetShowing('save-' + tok)) afterEdit(res, tok, false);
      endSave();
    }, e => { restoreL(snap); paintLeader(); notSaved(e.message || 'Check your signal and try again.'); endSave(); });
  };
}

// Filling an open spot: offer the later open times in the same spot (never ones someone already has)
function alsoBox(r, k, oldName, newName) {
  if (oldName || !newName) return '';
  const later = L.slots.map((s, j) => j).filter(j => j > k && isOpenCell(r, j));
  if (!later.length) return '';
  const rp = rosterOf(newName);
  return '<div class="alsobox"><p class="eyebrow" style="margin:0 0 4px">Also fill the rest of the day?</p>' +
    '<p class="small muted" style="margin:0 0 8px">These later times in ' + esc(r.spot) + ' are still open. Tick the ones ' + esc(firstLast(newName)) + ' should take too.</p>' +
    later.map(j => {
      const busy = rp && rp.busy[j] != null ? rp.busy[j] : '';
      return '<label class="radio"><input type="checkbox" data-also="' + j + '"' + (busy ? ' data-busy="1"' : '') + '> ' + esc(slotLabel(j, [r])) +
        (busy ? ' <span class="warn small">⚠ already at ' + esc(busy) + '</span>' : '') + '</label>';
    }).join('') +
    (later.length > 1 ? '<button class="btn sm line" id="alsoall" style="margin-top:6px">Tick all free times</button>' : '') + '</div>';
}

// The "Saved" window. While saving, its Text buttons stay locked so no one is texted about a change that didn't save.
function afterEdit(res, tok, saving) {
  const phoneOf = n => { const b = L.book[norm(n)] || (L.roster || []).filter(p => norm(p.name) === norm(n))[0]; return b ? b.phone : ''; };
  const last = n => String(n).split(',')[0].trim();
  const btn = (n, msg, label) => !phoneOf(n) ? '<p class="small muted">No phone on file for ' + esc(firstLast(n)) + '.</p>'
    : saving ? '<button class="btn" style="width:100%;margin-top:8px" disabled>⏳ ' + label + ' (unlocks when saved)</button>'
    : '<a class="btn" style="width:100%;margin-top:8px" href="' + smsHref(phoneOf(n), msg) + '">💬 ' + label + '</a>';
  let h = '<div id="save-' + tok + '"></div><div class="big">' + (saving ? '<span class="spin"></span>Saving…' : 'Saved ✓') + '</div><p class="muted small" style="margin:6px 0 10px">' + esc(res.spot) + ' · ' + esc(res.time) + '. ' +
    (saving ? 'The schedule already shows it. Please keep this page open for a few seconds.' : 'Everyone will see it when they refresh.') + '</p>';
  if (res.oldName) h += btn(res.oldName, 'Hey Brother ' + last(res.oldName) + ', your attendant assignment at ' + res.time + ' (' + res.spot + ') has been changed. Please refresh your schedule page to see your latest assignments: ' + personalLink(res.oldName), 'Text ' + firstLast(res.oldName));
  if (res.newName) h += btn(res.newName, 'Hey Brother ' + last(res.newName) + ', you have been assigned to ' + res.spot + ' at ' + res.time + '. Please refresh your schedule page to see your latest assignments: ' + personalLink(res.newName), 'Text ' + firstLast(res.newName));
  h += '<button class="btn line" style="width:100%;margin-top:12px" onclick="closeSheet()">Done</button>';
  openSheet(h);
}

// CI.who: { mode: 'leader' | 'checker', name, all, sections }
const CI = { people: [], status: {}, checkers: [], recent: {}, who: null, timer: null };

function ciInit(data, who) {
  if (!data) return;
  CI.people = data.people || []; CI.status = data.status || {}; CI.checkers = data.checkers || []; CI.who = who;
}
function ciCall(req) {
  req = Object.assign({ checkin: true }, req);
  return CI.who.mode === 'checker' ? call('apiVolunteer', CI.who.code, req) : call('apiEdit', L.code, req);
}
function ciStart(paint) { ciStop(); CI.timer = setInterval(() => ciCall({ op: 'poll' }).then(res => { if (res && res.ok) { ciTake(res); paint(); } }, () => {}), 10000); }
function ciStop() { if (CI.timer) { clearInterval(CI.timer); CI.timer = null; } }
function ciTake(res) { CI.status = res.status || {}; if (res.checkers) CI.checkers = res.checkers; }

// what the screen shows: saved status plus taps that are still saving (or just saved)
function ciState(key) {
  const r = CI.recent[key];
  if (r && (r.pending || Date.now() - r.t < 20000)) return r.st;
  return CI.status[key] || {};
}
function ciNowText() { const d = new Date(); return fmt(d.getHours() * 60 + d.getMinutes()); }
function ciTap(name, op, repaint, next) {
  const key = norm(name), cur = ciState(key), now = ciNowText(), me = CI.who.name;
  const st = op === 'in' ? Object.assign({}, cur, { in: cur.in || now, inBy: cur.inBy || me })
    : op === 'out' ? Object.assign({}, cur, { in: cur.in || now, inBy: cur.inBy || me, out: now, outBy: me, next: next || '' })
    : op === 'undoOut' ? { in: cur.in, inBy: cur.inBy } : {};
  CI.recent[key] = { st: st, pending: true, t: Date.now() };
  repaint();   // instant: no waiting for the server
  PENDING++;   // (warns before closing the page; no schedule refresh needed afterwards)
  ciCall({ op: op, name: name, next: next || '' }).then(res => {
    endSave(false);
    if (!res || !res.ok) { delete CI.recent[key]; repaint(); toast((res && res.error) || 'Not saved — try again'); return; }
    CI.recent[key] = { st: st, pending: false, t: Date.now() };
    ciTake(res); repaint();
  }, () => { endSave(false); delete CI.recent[key]; repaint(); toast('Not saved — check your signal and try again'); });
}

function ciLetter(name) { const c = String(name).trim().charAt(0).toUpperCase(); return c >= 'A' && c <= 'Z' ? c : 'Z'; }
// Split A–Z into one range per badge checker, as evenly as the last names allow
function ciRanges() {
  const k = CI.checkers.length;
  if (!k) return [];
  const counts = new Array(26).fill(0);
  CI.people.forEach(p => counts[ciLetter(p.name).charCodeAt(0) - 65]++);
  const total = CI.people.length, cum = []; let run = 0;
  counts.forEach((c, i) => { run += c; cum[i] = run; });
  const out = []; let start = 0;
  for (let g = 0; g < k; g++) {
    let end = 25;
    if (g < k - 1) {
      const target = total * (g + 1) / k, last = 25 - (k - 1 - g);
      end = start; let best = Infinity;
      for (let c = start; c <= last; c++) { const d = Math.abs(cum[c] - target); if (d < best) { best = d; end = c; } }
    }
    const from = String.fromCharCode(65 + start), to = String.fromCharCode(65 + end);
    out.push({ name: CI.checkers[g], from: from, to: to, label: from === to ? from : from + '–' + to,
      n: (cum[end] || 0) - (start ? cum[start - 1] : 0) });
    start = end + 1;
  }
  return out;
}
function ciMyRange() { return ciRanges().filter(r => norm(r.name) === norm(CI.who.name))[0] || null; }

// next time this person is needed (for the "not checked in" list)
function ciNeeded(p) {
  const now = nowInfo();
  if (!p.b.length) return { sort: 99999, text: 'No shifts scheduled', cls: '' };
  if (now !== null) {
    const on = p.b.filter(b => b[0] <= now && (b[1] == null || now < b[1]))[0];
    if (on) return { sort: -1, text: '⚠ On duty now: ' + on[2] + ' (since ' + fmt(on[0]) + ')', cls: 'late' };
    const nx = p.b.filter(b => b[0] > now)[0];
    return nx ? { sort: nx[0], text: 'Next: ' + fmt(nx[0]) + ' · ' + nx[2], cls: '' } : { sort: 99998, text: 'No more shifts today', cls: '' };
  }
  return { sort: p.b[0][0], text: 'First shift: ' + fmt(p.b[0][0]) + ' · ' + p.b[0][2], cls: '' };
}

function paintCheckin(root, onBack) {
  const who = CI.who, range = ciMyRange();
  // start on My letters for badge checkers, otherwise My sections for keymen, otherwise everyone
  if (UI.ciScope == null) UI.ciScope = range ? 'mine' : (!who.all && who.mode === 'leader' && (who.sections || []).length ? 'sections' : 'all');
  if (!UI.ciView) UI.ciView = 'all';
  let h = '';
  if (onBack) h += '<button class="btn sm line" id="ciback" style="margin-bottom:12px">← Back to my schedule</button>';
  h += '<div class="card"><p class="eyebrow">Check In / Check Out</p><div class="cistats" id="cistats"></div>' +
    (range ? '<p class="small" style="margin:10px 0 0">You check in last names <strong>' + esc(range.label) + '</strong>. You can still check in anyone.</p>' : '') + '</div>';
  if (who.all) h += ciCheckerCard();
  const scopes = [];
  if (range) scopes.push(['mine', 'My letters (' + range.label + ')']);
  if (!who.all && who.sections && who.sections.length && who.mode === 'leader') scopes.push(['sections', 'My sections']);
  if (scopes.length) scopes.push(['all', 'Everyone']);
  if (!scopes.some(s => s[0] === UI.ciScope)) UI.ciScope = 'all';
  const views = [['all', 'All'], ['missing', 'Missing Check-in'], ['in', 'Checked in'], ['out', 'Badge returned']];
  h += '<div class="card">' +
    (scopes.length ? '<div class="chips">' + scopes.map(s => '<button class="chip ' + (UI.ciScope === s[0] ? 'on' : '') + '" data-ciscope="' + s[0] + '">' + esc(s[1]) + '</button>').join('') + '</div>' : '') +
    '<div class="seg ciseg">' + views.map(v => '<button data-ciview="' + v[0] + '" class="' + (UI.ciView === v[0] ? 'on' : '') + '">' + v[1] + '</button>').join('') + '</div>' +
    '<input type="search" id="ciq" placeholder="Search names…" autocomplete="off" style="margin-top:10px"><div id="cirows" style="margin-top:8px"></div></div>';
  root.innerHTML = h;
  $('#ciq').value = UI.ciq || '';
  const repaint = () => ciRows();
  $('#ciq').oninput = () => { UI.ciq = $('#ciq').value; repaint(); };
  root.querySelectorAll('[data-ciscope]').forEach(b => b.onclick = () => { UI.ciScope = b.dataset.ciscope; paintCheckin(root, onBack); });
  root.querySelectorAll('[data-ciview]').forEach(b => b.onclick = () => { UI.ciView = b.dataset.ciview; paintCheckin(root, onBack); });
  $('#cirows').onclick = e => {
    const b = e.target.closest('[data-ciop]');
    if (!b) return;
    if (b.dataset.ciop === 'out') ciReturnAsk(b.dataset.ciname, repaint);
    else ciTap(b.dataset.ciname, b.dataset.ciop, repaint);
  };
  if (onBack) $('#ciback').onclick = () => { ciStop(); onBack(); };
  if (who.all) bindCheckerCard(root, onBack);
  repaint();
  ciStart(repaint);
}

function ciRows() {
  if (!$('#cirows')) { ciStop(); return; }
  const range = ciMyRange(), q = norm(UI.ciq || '');
  // the totals follow the chosen group (My letters / My sections / Everyone)
  const scoped = CI.people.filter(p => {
    if (UI.ciScope === 'mine' && range) { const c = ciLetter(p.name); if (c < range.from || c > range.to) return false; }
    if (UI.ciScope === 'sections') { const mine = (CI.who.sections || []).map(norm); if (!p.sec.some(s => mine.indexOf(norm(s)) >= 0)) return false; }
    return true;
  });
  let inN = 0, outN = 0;
  scoped.forEach(p => { const st = ciState(norm(p.name)); if (st.in) inN++; if (st.out) outN++; });
  $('#cistats').innerHTML = '<div><b>' + inN + '</b><span>Checked in</span></div><div class="miss"><b>' + (scoped.length - inN) + '</b><span>Not checked in</span></div><div><b>' + outN + '</b><span>Badges returned</span></div>';
  let list = scoped.filter(p => {
    if (q && !norm(p.name).includes(q) && !norm(firstLast(p.name)).includes(q)) return false;
    const st = ciState(norm(p.name));
    return UI.ciView === 'all' || (UI.ciView === 'missing' && !st.in) || (UI.ciView === 'in' && st.in && !st.out) || (UI.ciView === 'out' && st.out);
  });
  if (UI.ciView === 'missing') list = list.map(p => ({ p: p, need: ciNeeded(p) })).sort((a, b) => a.need.sort - b.need.sort).map(x => x.p);
  $('#cirows').innerHTML = list.map(p => {
    const st = ciState(norm(p.name)), parts = p.name.split(',');
    const nm = '<strong>' + esc(parts[0].trim()) + '</strong>' + (parts[1] ? ', ' + esc(parts[1].trim()) : '');
    let info, btns;
    const last = parts[0].trim(), text = msg => p.ph ? '<a class="btn sm xs ghost" href="' + smsHref(p.ph, msg) + '">Text</a>' : '';
    if (st.out) { info = '<span class="cires out">↩ Badge returned ' + esc(st.out) + (st.next ? ' · Next time: ' + esc(st.next) : '') + '</span>'; btns = '<button class="btn sm xs ghost" data-ciop="undoOut" data-ciname="' + esc(p.name) + '">Undo</button>'; }
    else if (st.in) { info = '<span class="cires in">✓ Checked in ' + esc(st.in) + (st.inBy ? ' · ' + esc(firstLast(st.inBy).split(' ')[0]) : '') + '</span>';
      btns = (UI.ciView === 'in' ? text('Hey brother ' + last + '. Our records show you still have a badge checked out. Please let us know if you have the badge still or if you turned it back in.') : '') +
        '<button class="btn sm xs line" data-ciop="out" data-ciname="' + esc(p.name) + '">Badge Returned</button><button class="btn sm xs ghost" data-ciop="undoIn" data-ciname="' + esc(p.name) + '">Undo</button>'; }
    else { const need = ciNeeded(p); info = '<span class="small ' + (need.cls === 'late' ? 'cilate' : 'muted') + '">' + esc(need.text) + '</span>';
      btns = text('Hey brother ' + last + '. Our records show you haven’t checked in and received a badge yet. Please come to the attendant desk to get your badge. If you already received a badge, just let us know and we will correct our records.') +
        '<button class="btn sm" data-ciop="in" data-ciname="' + esc(p.name) + '">Check in</button>'; }
    return '<div class="cirow' + (st.in ? ' done' : '') + '"><div class="grow">' + nm + '<div>' + info + '</div></div><div class="cibtns">' + btns + '</div></div>';
  }).join('') || '<p class="muted">' + (q ? 'No one matches that name.' : 'No one here.') + '</p>';
}

// "Badge Returned" asks whether he'd serve next time; Yes adds him to a tab on the Master Contact Sheet
function ciReturnAsk(name, repaint) {
  openSheet('<div class="big">Badge returned · ' + esc(firstLast(name)) + '</div><p style="margin:10px 0 14px">Is this brother willing to serve as an attendant next assembly?</p>' +
    '<div class="cinext"><button class="btn" data-next="Yes">Yes</button><button class="btn line" data-next="No">No</button><button class="btn line" data-next="Unsure">Unsure</button></div>' +
    '<button class="btn ghost" style="width:100%;margin-top:10px" onclick="closeSheet()">Cancel</button>');
  document.querySelectorAll('[data-next]').forEach(b => b.onclick = () => { closeSheet(); ciTap(name, 'out', repaint, b.dataset.next); });
}

// Overseers: choose the badge checkers (last names split between them in this order)
function ciCheckerCard() {
  const rs = ciRanges();
  let h = '<div class="card"><button class="collhead" id="cctoggle"><span class="grow"><span class="eyebrow" style="margin:0">Badge checkers</span><span class="small muted" style="display:block">' +
    (rs.length ? esc(rs.map(r => firstLast(r.name).split(' ')[0] + ' ' + r.label).join(' · ')) : 'No one assigned yet') + '</span></span><span class="plus">' + (UI.ccOpen ? '−' : '+') + '</span></button>';
  if (UI.ccOpen) {
    h += '<p class="small muted" style="margin:10px 0 8px">Last names are split as evenly as possible, in this order. Badge checkers see a Check In/Out button on their own schedule page.</p>' +
      rs.map((r, i) => '<div class="linkrow"><div class="grow"><strong>' + esc(firstLast(r.name)) + '</strong><div class="small muted">Last names ' + esc(r.label) + ' · ' + r.n + ' people</div></div>' +
        (i ? '<button class="btn sm xs line" data-ccup="' + i + '">↑</button>' : '') + '<button class="btn sm xs ghost" data-ccdel="' + i + '">Remove</button></div>').join('') +
      '<div class="row" style="gap:8px;margin-top:10px"><select id="ccadd" class="grow"><option value="">Add a badge checker…</option>' +
      CI.people.filter(p => !CI.checkers.some(c => norm(c) === norm(p.name))).map(p => '<option>' + esc(p.name) + '</option>').join('') + '</select>' +
      '<button class="btn sm" id="ccaddbtn">Add</button></div>' +
      '<button class="btn sm danger" id="ciclear" style="margin-top:14px">Clear all check-ins…</button>';
  }
  return h + '</div>';
}
function bindCheckerCard(root, onBack) {
  const again = () => paintCheckin(root, onBack);
  $('#cctoggle').onclick = () => { UI.ccOpen = !UI.ccOpen; again(); };
  if (!UI.ccOpen) return;
  const save = list => { CI.checkers = list; again(); ciCall({ op: 'setCheckers', checkers: list }).then(res => { if (res && res.ok) { ciTake(res); again(); } else toast((res && res.error) || 'Not saved'); }); };
  root.querySelectorAll('[data-ccdel]').forEach(b => b.onclick = () => save(CI.checkers.filter((c, i) => i !== Number(b.dataset.ccdel))));
  root.querySelectorAll('[data-ccup]').forEach(b => b.onclick = () => { const l = CI.checkers.slice(), i = Number(b.dataset.ccup); [l[i - 1], l[i]] = [l[i], l[i - 1]]; save(l); });
  $('#ccaddbtn').onclick = () => { const v = $('#ccadd').value; if (v) save(CI.checkers.concat([v])); };
  $('#ciclear').onclick = () => {
    openSheet('<div class="big" style="color:var(--warn)">⚠ Clear all check-ins?</div><p style="margin:10px 0 14px">This erases every check-in and check-out for everyone (for example, after testing). It can’t be undone.</p>' +
      '<div class="btns"><button class="btn danger" id="yesclear">Yes, clear all</button><button class="btn line" onclick="closeSheet()">Cancel</button></div>');
    $('#yesclear').onclick = () => { busy($('#yesclear'), 'Clearing…'); ciCall({ op: 'clearAll' }).then(res => { closeSheet(); if (res && res.ok) { CI.recent = {}; ciTake(res); toast('All check-ins cleared'); again(); } else toast((res && res.error) || 'Not cleared'); }); };
  };
}

/* ---------- new version: reload so no one keeps using an old copy of the page ---------- */
// Every 2 minutes, and whenever the page comes back on screen, ask the server for its version.
// If it changed, reload on our own when nothing is in progress; otherwise (or if the browser blocks it) show a bar to tap.
let NEWVER = false;
function checkBuild() {
  if (NEWVER || !BOOT.build) return;
  call('apiVolunteer', '', { build: true }).then(r => { if (r && r.build && r.build !== BOOT.build) newVersion(); }, () => {});
}
function newVersion() {
  NEWVER = true;
  const code = (L && L.code) || VCODE || BOOT.code;
  const url = code ? leaderLink(code) : (BOOT.shared ? BOOT.pageUrl + '?app=' + BOOT.app : BOOT.appUrl);
  const bar = document.createElement('a');
  bar.className = 'newver'; bar.href = url; bar.target = '_top';
  bar.textContent = '✨ This page was updated. Tap here to load the new version.';
  document.body.appendChild(bar);
  const a = document.activeElement;
  const busyNow = PENDING || $('#sheet').classList.contains('show') || (a && /INPUT|TEXTAREA|SELECT/.test(a.tagName));
  if (!busyNow) { try { window.top.location.href = url; } catch (e) { /* blocked: the bar stays */ } }
}
setInterval(checkBuild, 120000);
document.addEventListener('visibilitychange', () => { if (document.visibilityState === 'visible') checkBuild(); });
